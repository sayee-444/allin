"""A Server for manual tests (from issues/166/fallproxy.py): proxies port 19932 to the llama on
9932, but while <dir>/fall exists a POST reaches it and the connection is cut before any
answer (a fall, ADR-0023); while <dir>/down exists, /health answers 503.
#171: only a POST whose body holds the bytes written in <dir>/fall falls (every POST, if it is
empty), and each fall creates <dir>/down, so Prompts can be queued before the next try.
#174: unchanged from issues/171.
#176: while <dir>/hold exists, a POST whose body holds its bytes waits, unanswered, until it is
removed: its Turn stays in its HTTP request, as when Main stops before the Turn ends.
#185: while <dir>/crash exists, a POST whose body holds its bytes (once no longer held) gets
HTTP 400 back: its Step fails as a request the Server refused (LlmRequestFailed), not a fall."""
import json
import socket
import sys
import time
import urllib.request
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path

FALL = Path(sys.argv[1]) / "fall"
DOWN = Path(sys.argv[1]) / "down"  # while it exists, /health answers 503
HOLD = Path(sys.argv[1]) / "hold"  # #176: while it exists, a POST holding its bytes waits
CRASH = Path(sys.argv[1]) / "crash"  # #185: while it exists, a POST holding its bytes gets 400
REAL = "http://127.0.0.1:9932"


class Handler(BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.0"

    def log_message(self, *args): pass

    def _proxy(self, body=None):
        request = urllib.request.Request(REAL + self.path, data=body, method=self.command)
        for key in ("Content-Type", "Authorization", "Accept"):
            if self.headers.get(key):
                request.add_header(key, self.headers[key])
        try:
            reply = urllib.request.urlopen(request, timeout=600)
        except urllib.error.HTTPError as error:
            reply = error
        self.send_response(reply.status if hasattr(reply, "status") else reply.code)
        for key, value in reply.headers.items():
            if key.lower() not in ("transfer-encoding", "connection", "content-length"):
                self.send_header(key, value)
        self.end_headers()
        while chunk := reply.read1(4096):
            self.wfile.write(chunk); self.wfile.flush()

    def do_GET(self):
        if DOWN.exists() and self.path == "/health":  # the queue pauses until it is removed
            self.send_response(503); self.end_headers()
            return
        self._proxy()

    def do_POST(self):
        body = self.rfile.read(int(self.headers.get("Content-Length", 0)))
        measuring = self.path in ("/apply-template", "/tokenize")  # never falls
        while HOLD.exists() and not measuring and HOLD.read_bytes().strip() in body:
            time.sleep(0.5)  # #176: held while <dir>/hold names it
        if CRASH.exists() and not measuring and CRASH.read_bytes().strip() in body:
            with open(CRASH.parent / "falls.log", "a") as log: log.write(f"crash {self.path}\n")
            reply = json.dumps({"error": {"code": 400, "message": "refused by fallproxy (#185)",
                                          "type": "invalid_request_error"}}).encode()
            self.send_response(400); self.send_header("Content-Type", "application/json")
            self.send_header("Content-Length", str(len(reply))); self.end_headers()
            self.wfile.write(reply)
            return
        if FALL.exists() and not measuring and FALL.read_bytes().strip() in body:
            with open(FALL.parent / "falls.log", "a") as log: log.write(f"fall {self.path}\n")
            DOWN.touch()  # #171: the queue pauses after each fall, until DOWN is removed
            self.connection.shutdown(socket.SHUT_RDWR); self.close_connection = True
            return
        self._proxy(body)


ThreadingHTTPServer(("127.0.0.1", 19932), Handler).serve_forever()
